cmake_minimum_required(VERSION 3.24)
project(FloeChemistryBridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_QT OFF CACHE BOOL "" FORCE)
set(USE_OPENGL OFF CACHE BOOL "" FORCE)
set(USE_SPGLIB OFF CACHE BOOL "" FORCE)
set(USE_LIBMSYM OFF CACHE BOOL "" FORCE)
set(USE_LIBARCHIVE OFF CACHE BOOL "" FORCE)
set(USE_PLOTTER OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_FUZZ OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(USE_PYTHON OFF CACHE BOOL "" FORCE)
# Vendored snapshots must never inherit the enclosing Floe repository's git describe.
set(AVOGADRO_RELEASE_BUILD ON CACHE BOOL "" FORCE)
add_subdirectory(../Vendor/avogadrolibs avogadro)
add_library(FloeChemistryBridge STATIC bridge.cpp)
target_include_directories(FloeChemistryBridge PUBLIC include PRIVATE ../Vendor/avogadrolibs/thirdparty)
target_link_libraries(FloeChemistryBridge PRIVATE Avogadro::Core Avogadro::IO Avogadro::Calc Avogadro::QuantumIO)
